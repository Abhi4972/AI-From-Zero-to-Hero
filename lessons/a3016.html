In plain words first
The idea in one line

This repository contains a collection of practical examples demonstrating how to leverage Snowflake's AISQL capabilities across various business domains.

What it does

This repository contains a collection of practical examples demonstrating how to leverage Snowflake's AISQL capabilities across various business domains. Each subfolder contains a complete sample application with notebooks, supporting files, and use case details.

Marketing Lead Pipeline

Automated marketing lead screening and qualification

Demonstrate how Snowflake's marketing team uses AISQL to analyze, cleanse, and enrich incoming leads from web forms. The pipeline filters spam, scores leads on ICP fit, categorizes by seniority, and enables personalized outreach—accelerating sales cycles and increasing conversion rates.

E-commerce Customer Reviews

Customer feedback analysis and response automation

Process large volumes of unstructured customer reviews to identify product issues, analyze sentiment, and automate response generation. This solution helps e-commerce companies understand customer concerns, prioritize product improvements, and scale customer service operations.

Ad Campaign Analytics

Image-based advertising optimization

Transform ad campaign performance by analyzing image content alongside traditional metrics. This solution demonstrates image classification, content-based filtering, tagline generation, and sentiment analysis—all using familiar SQL syntax without switching between specialized tools.

CPG Promotion Management

AI-powered promotion and feedback analysis for consumer goods

Unlock ROI from trade promotions by extracting actionable insights from customer feedback, social posts, and call center notes using AISQL. This use case demonstrates how to: - Create synthetic tables for promotions and feedback - Filter and classify feedback with AISQL functions - Analyze sentiment and summarize campaign performance - Detect competitor activity and generate improved marketing copy

Retail and CPG teams can leverage these techniques to optimize promotional spend, boost compliance, and drive real-time campaign improvements—all directly in SQL.

import pandas as pd
from snowbook.executor import sql_executor
from snowbook.executor.sql_connector import ensure_session
import json
from functools import partial

def file_to_url(session, file_str):
    if file_str is None:
        return None
    file_obj = json.loads(file_str)
    url = session.sql(f"""select GET_PRESIGNED_URL('{file_obj["STAGE"]}', '{file_obj["RELATIVE_PATH"]}', 604800)""").collect()[0][0]
    return url

if "patched" not in st.session_state:
    import streamlit as st
    import pandas as pd
    from snowbook.executor import sql_executor

    old_sql_statement = sql_executor.run_single_sql_statement

    def patched_run_single_sql_statement(*args, **kwargs):
        result = old_sql_statement(*args, **kwargs)

        result_df = result.query_scan_data_frame
        image_columns = {}
        import streamlit as st

        for col, type in result_df.dtypes:
            if type == "file":
                image_columns[col] = st.column_config.ImageColumn()
        if image_columns:
            session = ensure_session()
            result_df_pd = result_df.to_pandas()
            for col in image_columns.keys():
                result_df_pd[col] = result_df_pd[col].apply(partial(file_to_url, session))
            st.dataframe(result_df_pd, column_config=image_columns)
        else:
            return result

    sql_executor.run_single_sql_statement = patched_run_single_sql_statement
    st.session_state.patched = True

/*--
• Create images table from uploaded files in the stage
--*/

CREATE OR REPLACE STAGE internal_stage_for_images
  COMMENT = 'Internal stage to store copied files from public S3'
  ENCRYPTION = (TYPE = 'SNOWFLAKE_SSE')
  DIRECTORY = (ENABLE = TRUE);

COPY FILES INTO @internal_stage_for_images
FROM @s3load
PATTERN = '.*\.png';

ALTER STAGE internal_stage_for_images REFRESH;

CREATE OR REPLACE TABLE images AS
SELECT
    TO_FILE(file_url) AS img_file,
    file_url,
    relative_path,
    last_modified
FROM
    directory(@internal_stage_for_images);


-- Setup completion note
SELECT 'Advertisement analytics setup is complete' AS note;

SELECT
    img_file,
    file_url,
    last_modified
FROM images;

-- Review the table structure and data types
DESCRIBE TABLE images;

SELECT
    relative_path AS image_name,
    AI_COMPLETE(
        'pixtral-large', 
        PROMPT(
            'Extract the text from this image and create an advertising tagline. Just return the tagline itself: {0}', 
            img_file
        )
    ) AS generated_tagline,
    file_url
FROM images
ORDER BY relative_path;

SELECT 
    relative_path AS image_name,
    img_file, 
    file_url, 
    last_modified 
FROM images
WHERE AI_FILTER(
    PROMPT(
        'Does this image {0} contain an espresso machine, coffee pot, french press, or other coffee brewing equipment?', 
        img_file
    )
)
ORDER BY relative_path;

CREATE OR REPLACE TABLE images_classified AS
SELECT
    img_file,
    SPLIT_PART(relative_path, '/', -1) AS file_name,
    AI_CLASSIFY(
        PROMPT(
            'Please help me classify the type of advertisement within this coffee-related image {0}', 
            img_file
        ), 
        [
            'long text', 
            'product shot', 
            'lifestyle', 
            'promo offer'
        ]
    ) AS classification_result,
    PARSE_JSON(classification_result):labels[0]::TEXT AS ad_type,
    file_url, 
    relative_path,
    last_modified
FROM images;

-- Review the classified advertisements
SELECT 
    file_name,
    ad_type,
    file_url
FROM images_classified
ORDER BY ad_type, file_name;

SELECT
    image_type,
    AI_AGG(
        'Image Type: ' || image_type || '\n\nFeedback: ' || impression_review, 
        'Score the sentiment of the feedback on a scale of 1 to 5 where 1 is very negative and 5 is very positive. Just write 1, 2, 3, 4, or 5. Return the average value only'
    ) AS sentiment_score,
    AI_AGG(
        'Image Type: ' || image_type || '\n\nFeedback: ' || impression_review, 
        'Analyze the feedback from these reviews and summarize the key insights in one concise sentence'
    ) AS feedback_summary 
FROM impressions_with_reviews
GROUP BY image_type
ORDER BY sentiment_score DESC;

-- Create a view of promotion-related feedback using AI_FILTER
CREATE OR REPLACE VIEW PROMO_FEEDBACK AS
SELECT *
FROM FEEDBACK
WHERE AI_FILTER(
  PROMPT('In the following text, does the customer mention a promotion, coupon, discount code, deal, offer, sale or bundle? Text: {0}', feedback_txt)
);

-- Inspect the filtered promo feedback
SELECT * FROM PROMO_FEEDBACK ORDER BY feedback_id;

-- Create a view that removes obvious spam or irrelevant comments
CREATE OR REPLACE VIEW PROMO_FEEDBACK_CLEAN AS
SELECT *
FROM PROMO_FEEDBACK
WHERE NOT AI_FILTER(
  PROMPT('Return TRUE if the comment is spam, unrelated to retail promotions or mentions real estate, gift cards or other scams. Comment: {0}', feedback_txt)
);

-- Inspect the cleaned promo feedback
SELECT *
FROM PROMO_FEEDBACK_CLEAN
ORDER BY feedback_id;

       AI_CLASSIFY(

           'output_mode': 'multi'
         }
       ):labels AS category,
       feedback_txt
FROM PROMO_FEEDBACK_CLEAN
ORDER BY feedback_id;
